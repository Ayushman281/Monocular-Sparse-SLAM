cmake_minimum_required(VERSION 3.16)
project(monocular_sparse_slam LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build mode" FORCE)
endif()
find_package(stella_vslam 0.7 REQUIRED CONFIG)
find_package(OpenCV 4 REQUIRED COMPONENTS core imgproc videoio)
find_package(Eigen3 REQUIRED)
add_executable(slam_runner src/main.cpp)
target_link_libraries(slam_runner PRIVATE stella_vslam::stella_vslam ${OpenCV_LIBS})
target_compile_options(slam_runner PRIVATE -Wall -Wextra -Wpedantic)
target_compile_definitions(slam_runner PRIVATE STELLA_COMMIT="525231147319bcd31242f981078c36d9272727b4")
set_target_properties(slam_runner PROPERTIES INSTALL_RPATH "$ORIGIN/../lib")
install(TARGETS slam_runner RUNTIME DESTINATION bin)
